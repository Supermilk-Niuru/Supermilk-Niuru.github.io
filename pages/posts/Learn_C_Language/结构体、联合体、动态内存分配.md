### 结构体的定义和调用
指针用`->`，变量用`.`

```c
#include<stdio.h>
#include<string.h>
struct person
{
    char name[100];
    char gender;
    int age;
    double height;
};
int main()
{
    struct person m1;
    m1.age = 3;
    strcpy(m1.name,"zhangsan");
    printf("%s %d\n",m1.name,m1.age);
}
```

1. 结构体相当于自己定义了一个数据类型，如本处的`struct person`。就相当于我们之前常用的`int`,`double`等等
2. 结构体中是有一批组合而成的数据。
3. <font style="color:#DF2A3F;">不要忘记最后的</font>`<font style="color:#DF2A3F;">;</font>`
4. `main`函数中的`struct person m1`,`m1`就是一个变量，其数据类型是`struct person`。类似于`int a`

### 结构体数组
+ **普通定义**

```c
#include<stdio.h>
struct student
{
    char name[100];
    int age;
    char gender;
    double height;
};
int main()
{
    struct student person1 = {"zhangsan",4,'M',1.78};
    struct student person2 = {"lisi",5,'M',1.79};
    struct student person3 = {"wangwu",6,'F',1.80};
    struct student a[3] = {person1,person2,person3};
   int i;
   for(i=0;i<3;i++)
   {
        printf("%s,%d,%c,%lf\n",a[i].name,a[i].age,a[i].gender,a[i].height);
   }
}
```

其实和之前定义数组一样，只不过把变量类型换成了`struct student`

由于`printf`一次性只能读取基本类型，不能直接输出整个结构体，所以要分开输出。

为了使代码变得更加简洁，我们可以修改一下`for`循环中的语句：

```c
for(i=0;i<3;i++)
   {
        struct student tmp = a[i];
        printf("%s,%d,%c,%lf\n",tmp.name,tmp.age,tmp.gender,tmp.height);
   }
```

+ **指针定义**

```c
#include<stdio.h>
struct student
{
    char name[100];
    int age;
    char gender;
    double height;
};
int main()
{
    struct student person1 = {"zhangsan",4,'M',1.78};
    struct student person2 = {"lisi",5,'M',1.79};
    struct student person3 = {"wangwu",6,'F',1.80};
    struct student *p[3] = {&person1,&person2,&person3};
    int i;
    for(i=0;i<3;i++)
    {
        printf("%s,%d,%c,%lf\n",p[i]->name, p[i]->age, p[i]->gender, p[i]->height);
    }
return 0;
```

`p[i]->name` 就是去“`p[i]` 所指向的那块内存地址”中，找到名为 `name` 的那个成员。`p[i]`中存储的是结构体的地址。

```c
for(i=0;i<3;i++)
    {
        struct student *s = p[i];
        printf("%s,%d,%c,%lf\n",s->name,s->age, s->gender, s->height);
    }
```

**<font style="color:#DF2A3F;">！！！</font>**`**<font style="color:#DF2A3F;">a[i].name</font>**`<font style="color:#DF2A3F;"> 等价于 </font>`**<font style="color:#DF2A3F;">p[i]->name</font>**`

###### 起别名
结构体前要加上关键字`typedef`

```c
#include<stdio.h>
typedef struct student
{
    char name[100];
    int age;
    char gender;
    double height;
} SD;
int main()
{
    SD person1 = {"zhangsan",40,'F',1.78};
    SD person2 = {"lisi",50,'M',1.79};
    SD person3 = {"wangwu",60,'F',1.80};
    SD a[3] = {person1,person2,person3};
    int i;
    for(i=0;i<3;i++)
    {
        SD tmp = a[i];
        printf("%s %d %c %lf\n",tmp.name,tmp.age,tmp.gender,tmp.height);
    }
    return 0;
}   
```

此时，`SD`就是结构体`struct student`的别名，后续就可以不必像之前那么繁琐写全名，只需要用`SD`即可(**<font style="color:#DF2A3F;">不要</font>**加`struct`)

结构体嵌套时，如果使用大括号赋值，也需要嵌套！

```c
//编写一个程序，统计A,B,C,D四个景点的投票数，决定最后去的景点，80个人投票。若票数相同，优先级A>B>C>D.
#include<stdio.h>
#include<stdlib.h>
#include<time.h>
/*
定义一个结构体，包含景点名和票数
生成随机数
统计票数
比较票数，得出最高票数的景点
相同票数的处理
输出结果
*/
struct Spots
{
    char name;
    int vote;
};

void count(struct Spots m1[4]);

int main()
{
    struct Spots m1[4]= {{'A',0},{'B',0},{'C',0},{'D',0}};
    count(m1);
    return 0;
}
void count(struct Spots m1[4])
{
    int i,k;
    srand(time(NULL)); 
    //统计票数
    for(i=0;i<80;i++)
    {
        int p = rand() % 4; 
        int choose = p%4;
        m1[choose].vote++;
    }
    //比较票数,找到最大票数
    int max = m1[0].vote;
    k = 0;
    for(i=1;i<=3;i++)
    {
        if(max < m1[i].vote)
        {
            max = m1[i].vote;
            k = i;
        }
        //相同票数的处理
        else if(max == m1[i].vote)
        {
            continue;
        }
    }
    //输出结果
    printf("票数最高的景点是%c,它的票数是%d\n",m1[k].name,m1[k].vote);
    for(i=0;i<=3;i++)
    {
        printf("景点%c,票数%d\n",m1[i].name,m1[i].vote);
    }
}
```

### 内存对齐
普通变量和结构体都有内存对齐。

###### 结构体的内存总大小
最大类型所占字节数的整数倍。

###### 每一个变量的位置
必须放在其类型大小的整数倍的内存地址上(**<font style="color:#DF2A3F;">CPU读取内存是整块整块地读，不是一个字节一个字节地读，这样可以节省时间</font>**)。由于补位产生的中间空缺的内存单元，并不是意味着上一个变量所占据的内存增加了。

**结论**：数据类型所占字节数从小到大排列(<font style="color:#117CEE;">如果有相同字节的，排在一起</font>)，可以节约空间，避免浪费。

### 联合体(共用/同体)
一个数据可能有多种类型去表示，我们就要使用联合体`union`

###### 定义和调用联合体
联合体的定义与调用和结构体的定义与调用类似，起别名也是类似的。

**<font style="color:#DF2A3F;">在赋值时每次只能赋一个值！！！第二次赋值会覆盖原有的数据。</font>**

```c
union Moneytype
{
    int money1;
    double money2;
    char money3[100];
}；
int main()
{
    union Moneytype m1;
    m1.money1 = 100;
    printf("%d\n",m1.money1);
    return 0;
}
```

```c
typedef Moneytype
{
    int money1;
    double money2;
    char money3[100];
}Mt;
int main()
{
    M m1;
    m1.money1 = 100;
    printf("%d\n",m1.money1);
    return 0;
}
```

###### 占用内存(注意内存对齐)
$ 所占内存的大小=联合体中最大成员占用的内存大小(且总大小是最大成员的整数倍) $

**注意**：

`char money3[100]`是按照$ 100 $个`char`类型存储。

`m1`所占字节数是$ 104 $。

### 动态内存分配
###### 常用函数
`malloc`：申请**<font style="color:#DF2A3F;">连续</font>**空间。

`calloc`：申请空间并且数据初始化。

`realloc`：改写内存空间大小。

`free`：释放内存空间。

```c
#include<stdio.h>
#include<stdlib.h>
#include<string.h>
int main()
{
    //int *p = malloc(sizeof(int)*10);
    int *p = calloc(10,sizeof(int));
    //赋值
    int i;
    for(i=0;i<10;i++)
    {
        *(p+i) = (i+1)*10;
        //printf("%d ",p[i]);
    }
    //内存扩容
    int *pp = realloc(p,sizeof(int)*20);
    for(i=0;i<20;i++)
    {
        //*(p+i) = (i+1)*10;
        printf("%d ",pp[i]);
    }
    //printf("%d ",pp[0]);
    free(pp);
    return 0;
    
}
```

**注意**：

1. 避免不同编译环境的影响，使用`sizeof()`来表示申请的内存大小，使代码更有普适性。
2. 一般常使用`malloc`函数，效率更高，如果未赋初始值，则输出的是残留值。
3. 内存扩容之后，赋值，输入、输出**<font style="color:#DF2A3F;">不要用原来的指针</font>**，因为原来的指针的内存已经被释放，成为了野指针(😜<font style="color:#117CEE;">当然，也有特殊情况，原地址后面空间足够，</font>`<font style="color:#117CEE;">realloc</font>`<font style="color:#117CEE;">会直接在后面添加，新旧指针的地址完全一样</font><font style="color:#117CEE;">😜</font>)。最后的内存释放**<font style="color:#DF2A3F;">使用扩容之后的新指针</font>**，如果再次使用原来的指针，就会出现二次释放，编译器会强制停止运行(`zsh: abort`)。
